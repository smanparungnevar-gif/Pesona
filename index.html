Integrasi Google Sheets Backend

Hubungkan web Quick Count ini ke Google Spreadsheet agar data tersimpan permanen dan dapat diakses bersama secara gratis!

1

Buat Spreadsheet

Buka sheets.new lalu klik menu Ekstensi > Apps Script.

2

Paste Kode Script

Hapus isi file Code.gs, salin dan tempel kode script di bawah ini, lalu simpan.

3

Deploy Sebagai Web App

Klik Deploy > New Deployment. Pilih tipe Web App. Atur Who has access: Anyone. Salin Web App URL-nya.

*Setelah disambungkan, penambahan suara dari web akan otomatis masuk ke Google Spreadsheet secara realtime!

Kode Backend: Code.gs

/* ============================================================
 * GOOGLE APPS SCRIPT BACKEND (Code.gs)
 * Quick Count Pemilihan OSIS & MPK
 * ============================================================ */

function doGet(e) {
  var action = e.parameter.action;
  if (action === "getData") {
    return ContentService.createTextOutput(JSON.stringify(getQuickCountData()))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput("Quick Count API Running");
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000); // Mencegah bentrokan data bersamaan
  
  try {
    var data = JSON.parse(e.postData.contents);
    if (data.action === "addVote") {
      var result = recordVote(data);
      return ContentService.createTextOutput(JSON.stringify(result))
        .setMimeType(ContentService.MimeType.JSON);
    } else if (data.action === "initSheets") {
      setupSheets();
      return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Sheet Otomatis Dibuat!"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

// Menyiapkan Sheet dan Header secara otomatis
function setupSheets() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // Sheet Log Transaction
  var sheetLog = ss.getSheetByName("Log_Suara") || ss.insertSheet("Log_Suara");
  if (sheetLog.getLastRow() === 0) {
    sheetLog.appendRow(["Waktu", "Kategori", "Kandidat ID", "Nama Kandidat", "TPS", "Jumlah Suara"]);
    sheetLog.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#e2e8f0");
  }
  
  // Sheet Rekap Suara
  var sheetRekap = ss.getSheetByName("Rekap_Suara") || ss.insertSheet("Rekap_Suara");
  if (sheetRekap.getLastRow() === 0) {
    sheetRekap.appendRow(["Kategori", "ID", "Nama Paslon/Kandidat", "Total Suara"]);
    sheetRekap.getRange(1, 1, 1, 4).setFontWeight("bold").setBackground("#cbd5e1");
    
    // Default Paslon Data
    sheetRekap.appendRow(["OSIS", 1, "01. Andi & Budi", 0]);
    sheetRekap.appendRow(["OSIS", 2, "02. Citra & Dewi", 0]);
    sheetRekap.appendRow(["MPK", 1, "01. Eka", 0]);
    sheetRekap.appendRow(["MPK", 2, "02. Farhan", 0]);
    sheetRekap.appendRow(["MPK", 3, "03. Gita", 0]);
  }
}

// Rekam Suara Baru ke Sheet
function recordVote(payload) {
  setupSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetLog = ss.getSheetByName("Log_Suara");
  var sheetRekap = ss.getSheetByName("Rekap_Suara");
  
  var timestamp = new Date();
  sheetLog.appendRow([timestamp, payload.category, payload.candidateId, payload.candidateName, payload.tps, payload.votes]);
  
  // Update Total Suara di Sheet Rekap
  var data = sheetRekap.getDataRange().getValues();
  for (var i = 1; i < data.length; i++) {
    var kat = data[i][0];
    var id = data[i][1];
    if (kat === payload.category && id == payload.candidateId) {
      var currentVote = Number(data[i][3]) || 0;
      sheetRekap.getRange(i + 1, 4).setValue(currentVote + payload.votes);
      break;
    }
  }
  
  return { status: "success", data: getQuickCountData() };
}

// Ambil Data Terkini
function getQuickCountData() {
  setupSheets();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetRekap = ss.getSheetByName("Rekap_Suara");
  var sheetLog = ss.getSheetByName("Log_Suara");
  
  var dataRekap = sheetRekap.getDataRange().getValues();
  var result = { osis: [], mpk: [], logs: [] };
  
  for (var i = 1; i < dataRekap.length; i++) {
    var item = {
      id: Number(dataRekap[i][1]),
      nama: dataRekap[i][2],
      suara: Number(dataRekap[i][3]) || 0
    };
    if (dataRekap[i][0] === "OSIS") {
      result.osis.push(item);
    } else if (dataRekap[i][0] === "MPK") {
      result.mpk.push(item);
    }
  }
  
  // Ambil 10 Log Terakhir
  var dataLog = sheetLog.getDataRange().getValues();
  var logsCount = dataLog.length;
  for (var j = Math.max(1, logsCount - 15); j < logsCount; j++) {
    result.logs.push({
      waktu: dataLog[j][0],
      category: dataLog[j][1],
      candidateName: dataLog[j][3],
      tps: dataLog[j][4],
      votes: dataLog[j][5]
    });
  }
  
  return result;
}
